let bzCalls = 0;
async function bzGet(path: string) {
  bzCalls++;
  const r = await fetch(BZ + path, { headers: { Authorization: "Token " + BZ_KEY, Accept: "application/json" } });
  { const h = r.headers.get("ratelimit") || ""; const mm = /remaining=(\d+)/.exec(h); bzRem = mm ? mm[1] : (r.headers.get("ratelimit-remaining") ?? r.headers.get("x-ratelimit-remaining") ?? bzRem); }

  if (r.status === 429) throw new Error("Bzzoiro: limite di richieste raggiunto (rateLimit)");
  if (!r.ok) throw new Error("Bzzoiro " + r.status + " su " + path.split("?")[0]);

}
async function bzList(path: string, maxPages = 6) {
  let out: any[] = [];
  let url: string | null = path + (path.includes("?") ? "&" : "?") + "limit=200";
  for (let i = 0; i < maxPages && url; i++) {
    const d = await bzGet(url);
    const res = Array.isArray(d) ? d : (d.results || d.events || d.data || []);
    out = out.concat(res);
    const nx = Array.isArray(d) ? null : d.next;
    url = nx ? String(nx).replace(BZ, "") : null;
  }
  return out;
}
const num = (v: any) => (v == null || v === "" ? null : (isFinite(Number(v)) ? Number(v) : null));
const nm = (v: any) => (v == null ? "" : typeof v === "object" ? String(v.name ?? v.short_name ?? v.title ?? "") : String(v));
function bzTeam(e: any, side: "home" | "away") {
  return nm(e[side + "_team_obj"]) || nm(e[side + "_team"]) || nm(e[side]) || nm(e.teams && e.teams[side]);

function bzLeague(e: any) {
  const l = e.league ?? e.tournament ?? e.competition;
  if (l && typeof l === "object") return { name: nm(l), country: nm(l.country ?? l.category ?? "") };
  return { name: nm(l ?? e.league_name ?? ""), country: nm(e.country ?? e.league_country ?? "") };
}
function bzPeriod(e: any) {
  const p = String(e.period ?? e.status_period ?? "").toUpperCase(), st = String(e.status ?? "").toLowerCase();
  if (/cancel|postpon|abandon|suspend|interrupt/.test(st)) return /postpon/.test(st) ? "PST" : "CANC";
  if (p === "HT" || /half.?time|^ht$/.test(st)) return "HT";
  if (p === "FT" || /finish|ended|^ft$|after/.test(st)) return "FT";
  if (p === "2T" || p === "2H" || /second/.test(st)) return "2H";
  if (p === "1T" || p === "1H" || /first/.test(st)) return "1H";
  if (/live|inprogress|in_progress/.test(st)) return (num(e.current_minute) ?? 0) > 45 ? "2H" : "1H";
  return "NS";
}
/* punteggio del 1° tempo: cerca le varianti di nome più comuni */
function bzHT(e: any): number[] | null {
  const pairs = [["home_score_ht", "away_score_ht"], ["ht_home_score", "ht_away_score"], ["home_ht_score", "away_ht_score"],
    ["home_score_1h", "away_score_1h"], ["home_halftime_score", "away_halftime_score"], ["home_score_half", "away_score_half"]];
  for (const [a, b] of pairs) if (num(e[a]) != null && num(e[b]) != null) return [num(e[a])!, num(e[b])!];
  for (const k of ["halftime", "half_time", "ht", "ht_score", "score_ht", "first_half"]) {
    const v = e[k] ?? (e.scores && e.scores[k]) ?? (e.score && e.score[k]);
    if (v && typeof v === "object" && num(v.home) != null && num(v.away) != null) return [num(v.home)!, num(v.away)!];
    if (typeof v === "string" && /^\d+\s*[-:]\s*\d+$/.test(v)) return v.split(/[-:]/).map((x) => Number(x.trim()));
  }
  return null;
}

function bzGoals(inc: any[]) {

}
function bzReds(inc: any[]) {

function bzSide(i: any) {
  if (typeof i.is_home === "boolean") return i.is_home ? 0 : 1;
  if (i.is_home === 1 || i.is_home === 0) return i.is_home ? 0 : 1;
  const s = String(i.team ?? i.side ?? "").toLowerCase();

}
function bzStats(e: any) {
  const ls = e.live_stats ?? e.stats ?? (e.statistics && e.statistics.stats) ?? null;
  const st = ls && (ls.home || ls.away) ? ls : (ls && ls.stats) || null;
  const out: any = { s: 0, t: 0, c: 0, red: 0, xg: null as number | null, side: [{}, {}] as any[] };
  if (!st) return out;
  let xg = 0, hasXg = 0;
  (["home", "away"] as const).forEach((k, i) => {
    const h = st[k] || {};
    const xv = h.xg && typeof h.xg === "object" ? num(h.xg.actual ?? h.xg.value) : num(h.xg ?? h.expected_goals);
    const xl = num(e[k + "_xg_live"]);
    const S: any = {
      s: num(h.total_shots ?? h.shots), t: num(h.shots_on_target ?? h.shots_on_goal),
      b: num(h.shots_inside_box ?? h.shots_insidebox ?? h.inside_box), c: num(h.corner_kicks ?? h.corners),
      red: num(h.red_cards) ?? 0, poss: num(h.ball_possession ?? h.possession), xg: xv ?? xl,
    };
    if (S.xg != null) { xg += S.xg; hasXg++; }
    out.side[i] = S;
    out.s += S.s || 0; out.t += S.t || 0; out.c += S.c || 0; out.red += S.red || 0;
  });
  if (hasXg === 2) out.xg = Math.round(xg * 100) / 100; else out.side.forEach((x: any) => (x.xg = null));
  return out;
}

function bzLastGoal(inc: any[]) {
  const g = bzGoals(inc).map((i) => { const t = bzIncMinute(i); return t.m + t.x; });

function bzFirstGoal(inc: any[]) {
  const g = bzGoals(inc).map(bzIncMinute).sort((a, b) => a.m - b.m || a.x - b.x);
  if (!g.length) return null;
  return g[0].m + (g[0].m === 45 && g[0].x ? g[0].x : 0);
}
function bzHTfromInc(inc: any[]): number[] | null {
  const g = bzGoals(inc);

  for (const i of g) {

    const first = per != null ? per === 1 : t.m <= 45;

  return ht;
}

/* quote Over/Under di un evento: cerca le forme più comuni (campi odds_over_25 o liste di mercati) */
function bzParseOU(d: any) {
  const ou: Record<string, { o?: number; u?: number }> = {};
  const put = (line: string, side: "o" | "u", v: any) => { const n = num(v); if (n && n > 1) (ou[line] = ou[line] || {})[side] = n; };
  const walk = (x: any, depth: number) => {
    if (!x || depth > 4) return;
    if (Array.isArray(x)) { x.forEach((y) => walk(y, depth + 1)); return; }
    if (typeof x !== "object") return;
    for (const [k, v] of Object.entries(x)) {
      const m = /^odds_(over|under)_(\d)(\d)$/.exec(k);
      if (m) put(m[2] + "." + m[3], m[1] === "over" ? "o" : "u", v);
    }
    const name = String(x.market ?? x.name ?? x.market_name ?? "").toLowerCase();
    const line = num(x.line ?? x.handicap ?? x.total ?? x.point);
    if (/over|under|total|goals/.test(name) && line != null && !/half|team|corner|card/.test(name)) {
      if (x.over != null) put(line.toFixed(1), "o", x.over);
      if (x.under != null) put(line.toFixed(1), "u", x.under);
    }
    const sel = String(x.selection ?? x.outcome ?? x.label ?? "");
    const sm = /^(over|under)\s*(\d+(?:\.\d)?)$/i.exec(sel.trim());
    if (sm && !/half|team|corner|card/.test(name)) put(Number(sm[2]).toFixed(1), /over/i.test(sm[1]) ? "o" : "u", x.odd ?? x.odds ?? x.price ?? x.value);
    for (const v of Object.values(x)) if (v && typeof v === "object") walk(v, depth + 1);
  };
  walk(d, 0);
  return ou;
}

/* quota live del book per "altro gol entro la fine" (Over punteggio+0,5).
   Come nell'app: si usa solo se è sicuramente live (la linea già vinta non deve essere più quotata). */
async function bookN1(afId: string, G: number) {
  if (G < 1 || String(afId).startsWith("tc:")) return null;
  try {
    const raw = String(afId).replace(/^bz:/, "");
    const ou = bzParseOU(await bzGet(`/api/v2/events/${raw}/odds/`));
    const prev = ou[(G - 0.5).toFixed(1)], cur = ou[(G + 0.5).toFixed(1)];
    if (prev && prev.o && prev.o > 1.01) return null;
    return cur && cur.o && cur.o > 1 ? cur.o : null;
  } catch { return null; }
}

/* ===================== Pronostici prepartita automatici ===================== */
const PRE_EVERY = 6 * 3600e3;
function poisTail(need: number, L: number) { let p = 0, t = Math.exp(-L); for (let k = 0; k < need; k++) { p += t; t *= L / (k + 1); } return 1 - p; }
function devig(a?: number | null, b?: number | null) { return a && b && a > 1 && b > 1 ? (1 / a) / (1 / a + 1 / b) : null; }
/* gol attesi totali da una linea Over/Under: λ tale che P(almeno need gol) = probabilità senza aggio */
function lamFromLine(line: number, o?: number | null, u?: number | null) {
  const p = devig(o, u);
  if (p == null) return null;
  const need = Math.round(line + 0.5);
  let lo = 0.05, hi = 9;
  for (let i = 0; i < 50; i++) { const mid = (lo + hi) / 2; if (poisTail(need, mid) < p) lo = mid; else hi = mid; }
  return (lo + hi) / 2;
}
function pois(k: number, l: number) { let p = Math.exp(-l); for (let i = 1; i <= k; i++) p *= l / i; return p; }
function hda(lh: number, la: number) {
  let h = 0, d = 0, a = 0;
  for (let i = 0; i < 11; i++) for (let j = 0; j < 11; j++) { const v = pois(i, lh) * pois(j, la); if (i > j) h += v; else if (i === j) d += v; else a += v; }
  return { h, d, a };
}
function slugT(s: string) { return String(s).normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, ""); }
function romeParts(ts: number) {
  const p = Object.fromEntries(new Intl.DateTimeFormat("en-CA", { timeZone: "Europe/Rome", year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hourCycle: "h23" }).formatToParts(new Date(ts)).map((x) => [x.type, x.value]));
  return { date: `${p.year}-${p.month}-${p.day}`, time: `${p.hour}:${p.minute}` };
}
function preMatch(e: any) {
  const st = String(e.status ?? "").toLowerCase();
  if (st && !/notstarted|not_started|scheduled|^ns$/.test(st)) return null;
  const ts = Date.parse(e.event_date ?? e.start_time ?? e.date);
  if (!isFinite(ts) || ts < Date.now() + 5 * 60e3) return null;
  const home = bzTeam(e, "home"), away = bzTeam(e, "away");
  if (!home || !away) return null;
  const lines: number[] = [];
  for (const [ln, o, u] of [[1.5, e.odds_over_15, e.odds_under_15], [2.5, e.odds_over_25, e.odds_under_25], [3.5, e.odds_over_35, e.odds_under_35]] as any) {
    const l = lamFromLine(ln, num(o), num(u)); if (l) lines.push(l);
  }
  const p1x = [num(e.odds_home), num(e.odds_draw), num(e.odds_away)];
  const has1x2 = p1x.every((x) => x && x > 1);
  let L: number | null = lines.length ? Math.exp(lines.reduce((s, x) => s + Math.log(x), 0) / lines.length) : null;
  const pr = e._pred;
  if (!L && pr && num(pr.expected_home_goals) != null && num(pr.expected_away_goals) != null) L = num(pr.expected_home_goals)! + num(pr.expected_away_goals)!;
  if (!L) return null;
  /* divisione casa/ospite: la stessa differenza vittoria casa − vittoria ospite delle quote 1X2 */
  let s = 0.53, mk: any = null;
  if (has1x2) {
    const inv = p1x.map((x) => 1 / x!), S = inv[0] + inv[1] + inv[2], ph = inv[0] / S, pd = inv[1] / S, pa = inv[2] / S, target = ph - pa;
    let lo = 0.08, hi = 0.92;
    for (let i = 0; i < 40; i++) { const mid = (lo + hi) / 2, r = hda(L * mid, L * (1 - mid)); if (r.h - r.a < target) lo = mid; else hi = mid; }
    s = (lo + hi) / 2;
    mk = { p1: Math.round(ph * 1000) / 10, px: Math.round(pd * 1000) / 10, p2: Math.round(pa * 1000) / 10 };
  } else if (pr && num(pr.expected_home_goals) != null && num(pr.expected_away_goals) != null) {
    const a = num(pr.expected_home_goals)!, b = num(pr.expected_away_goals)!; if (a + b > 0) s = a / (a + b);
  }
  const lh = Math.round(L * s * 1000) / 1000, la = Math.round(L * (1 - s) * 1000) / 1000;
  const o15 = num(e.odds_over_15), o25 = num(e.odds_over_25), gg = num(e.odds_btts_yes);
  const odds: any = {};
  if (has1x2) { odds["1"] = p1x[0]; odds["X"] = p1x[1]; odds["2"] = p1x[2]; }
  if (o15) odds.o15 = o15; if (o25) odds.o25 = o25; if (gg) odds.gg = gg;
  if (mk) { const q15 = devig(o15, num(e.odds_under_15)), q25 = devig(o25, num(e.odds_under_25)); if (q15) mk.o15 = Math.round(q15 * 1000) / 10; if (q25) mk.o25 = Math.round(q25 * 1000) / 10; }
  const lg = bzLeague(e), rp = romeParts(ts);
  const m: any = {
    id: rp.date + "_" + slugT(home) + "_" + slugT(away), date: rp.date, time: rp.time,
    league: (lg.country ? lg.country + " · " : "") + lg.name, home, away, lh, la,
    rel: lines.length >= 2 && has1x2 ? "B" : "C", review: "auto",
    note: "Automatico: quote medie dei bookmaker (Bzzoiro)" + (lines.length ? "" : " · gol attesi dal loro modello"),
    src: lines.length ? "quote" : "modello Bzzoiro", lhM: lh, laM: la, wMkt: 1, afId: "bz:" + e.id,
    odds: Object.keys(odds).length ? odds : undefined, mkt: mk || undefined,
    fav: mk ? Math.round(Math.max(mk.p1, mk.p2)) : undefined,
  };
  if (!o15 && !o25) m.topSkip = "quote Over non ancora aperte";
  return m;
}
async function buildPrematch() {

  const evs = await bzList(`/api/v2/events/?date_from=${from}&date_to=${to}`, 20);
  /* previsioni di Bzzoiro (gol attesi): servono solo dove mancano le quote Over/Under */
  try {
    const preds = await bzList(`/api/predictions/?date_from=${from}&date_to=${to}`, 12);
    const byEv = new Map<any, any>();
    for (const p of preds) { const id = p.event?.id ?? p.event_id ?? p.event; if (id != null) byEv.set(String(id), p); }
    for (const e of evs) { const p = byEv.get(String(e.id)); if (p) e._pred = p; }
  } catch { /* facoltative */ }
  const matches = evs.map(preMatch).filter(Boolean);

}
/* TotalCorner: le partite dei prossimi giorni arrivano a pagine da 30 (circa 1.300 al giorno).
   Per stare nel limite di 30 richieste al minuto si leggono poche pagine per giro (ogni 2 minuti):
   oggi, domani e dopodomani si completano in circa un'ora (4 pagine per giro), poi si ricomincia ogni 6 ore. */
function tcPreMatch(x: any) {
  if (tcIsSim(x)) return null;
  const f = tcFixture(x);
  if (!f.home || !f.away || f.status !== "NS" || !f.kickoff) return null;
  const ts = Date.parse(f.kickoff);
  if (ts < Date.now() + 5 * 60e3) return null;
  const L = f.odds?.pre?.lam;
  if (!L) return null;
  const p1x = f.odds?.pre?.x12;
  let s = 0.53, mk: any = null;
  if (p1x) {
    const inv = p1x.map((v: number) => 1 / v), S = inv[0] + inv[1] + inv[2], ph = inv[0] / S, pd = inv[1] / S, pa = inv[2] / S, target = ph - pa;
    let lo = 0.08, hi = 0.92;
    for (let i = 0; i < 40; i++) { const mid = (lo + hi) / 2, r = hda(L * mid, L * (1 - mid)); if (r.h - r.a < target) lo = mid; else hi = mid; }
    s = (lo + hi) / 2;
    mk = { p1: Math.round(ph * 1000) / 10, px: Math.round(pd * 1000) / 10, p2: Math.round(pa * 1000) / 10 };
  }
  const lh = Math.round(L * s * 1000) / 1000, la = Math.round(L * (1 - s) * 1000) / 1000;
  /* si tengono solo le partite che possono entrare nei pronostici (Over 1.5 o gol nel 1° tempo plausibili) */
  const p15 = 1 - Math.exp(-L) * (1 + L), pht = 1 - Math.exp(-0.45 * L);
  /* 4.2.8 — i campionati di punta restano sempre (sezione "Top dei top campionati"), anche con pochi gol attesi */
  if (p15 < 0.72 && pht < 0.66 && !tipLeagueTop({ league: f.league, home: f.home, away: f.away })) return null;
  const rp = romeParts(ts);
  const odds: any = { gl: f.odds.pre.line };
  /* 4.0.4 — linea gol prepartita del 1° tempo (per studiare i filtri "gol nel 1° tempo" dal prepartita) */
  const glh = tcLine(x.p_goal_h ?? x.p_goal_half ?? x.p_hgoal);
  if (glh != null) odds.glh = glh;
  if (p1x) { odds["1"] = p1x[0]; odds["X"] = p1x[1]; odds["2"] = p1x[2]; }
  const bt = tcNum(Array.isArray(x.p_btts) ? x.p_btts[0] : x.p_btts);
  if (bt && bt > 1) odds.gg = bt;
  return {
    id: rp.date + "_" + slugT(f.home) + "_" + slugT(f.away), date: rp.date, time: rp.time, league: f.league, home: f.home, away: f.away, lh, la,
    rel: p1x ? "B" : "C", review: "auto", note: "Automatico: quote bet365 (TotalCorner), linea gol " + f.odds.pre.line,
    src: "quote", lhM: lh, laM: la, wMkt: 1, afId: f.afId, odds, mkt: mk || undefined, fav: mk ? Math.round(Math.max(mk.p1, mk.p2)) : undefined,
  };
}
async function tcPrematchStep(force = false) {
  if (tcOff == null) { const sv: any = await getState("tcOff"); if (sv != null) tcOff = Number(sv); }
  const st: any = (await getState("tcpre")) || {};
  const now = Date.now();
  const days = [0, 1, 2].map((n) => romeParts(now + n * 86400e3).date);
  /* 3.95.1 — il giro riparte: a mezzanotte (oggi e domani subito aggiornati), 6 ore dopo un giro COMPLETO, oppure dopo
     18 ore se si è bloccato. Prima ripartiva ogni 6 ore anche a metà: se il giro era lento non arrivava mai a domani.
     Le partite dei giorni già letti restano finché il nuovo giro non li rilegge. */
  if (force || !st.cycleAt || st.today !== days[0] || (st.done && now - st.cycleAt > PRE_EVERY) || (!st.done && now - st.cycleAt > 3 * PRE_EVERY) || !days.includes(st.day)) {
    st.cycleAt = now; st.today = days[0]; st.day = days[0]; st.page = 1; st.byDay = st.byDay || {}; st.done = false;
  }
  if (st.done) return st;
  /* 3.95.1 — con poche partite live (notte, mattina) più pagine per giro: oggi + domani pronti in ~20-30 minuti */
  let budget = Math.min((liveCache?.length ?? 0) < 40 ? 10 : 6, tcLeft());
  while (budget > 0 && !st.done) {
    const ymd = st.day.replace(/-/g, "");
    const r = await tcList("/match/schedule", { date: ymd, columns: "odds,goalLine,goalLineHalf,btts" }, Math.min(budget, 12), st.page); /* 4.0.4 — anche la linea gol del 1° tempo */
    budget -= Math.max(1, r.nextPage - st.page);
    /* 3.99.9 — una partita con dati strani non blocca più tutto il giro */
    const keep = r.list.map((x: any) => { try { return tcPreMatch(x); } catch (e) { console.error("prepartita", x?.id, (e as Error).message); return null; } }).filter(Boolean);
    const cur = st.byDay[st.day] && st.page > 1 ? st.byDay[st.day] : [];
    st.byDay[st.day] = cur.concat(keep);
    if (r.nextPage > r.pages || !r.list.length) {
      const k = days.indexOf(st.day);
      if (k < days.length - 1) { st.day = days[k + 1]; st.page = 1; } else st.done = true;
    } else st.page = r.nextPage;
  }
  for (const d of Object.keys(st.byDay)) if (!days.includes(d)) delete st.byDay[d];
  await setState("tcpre", st);
  if (st.done || Object.keys(st.byDay).length) {
    const seen = new Set<string>();
    const matches = days.flatMap((d) => st.byDay[d] || []).filter((m: any) => !seen.has(m.id) && (seen.add(m.id), true));
    const file = { app: "bet_tracker", kind: "pronostici", auto: true, batch: "auto-tc-" + new Date(st.cycleAt).toISOString().slice(0, 13) + (st.done ? "" : "-p" + matches.length),
      generatedAt: new Date(now).toISOString(), label: "Automatico · bet365 (TotalCorner) · " + romeParts(now).date + " " + romeParts(now).time + (st.done ? "" : " · in completamento"),
      from: days[0], to: days[2], matches };
    await setState("prematch", file);
  }
  return st;
}
async function getPrematch(force = false) {
  if (TC_TOKEN) {
    const cur: any = await getState("prematch");
    /* l'app legge solo quello che il giro ha già preparato: niente richieste in più (limite 30 al minuto) */
    if (cur && String(cur.batch || "").startsWith("auto-tc")) return cur;
    return { matches: [], error: "Pronostici automatici in preparazione: il giro ogni 2 minuti li completa in circa un'ora." };
  }
  const cur: any = await getState("prematch");
  if (!force && cur && cur.generatedAt && Date.now() - Date.parse(cur.generatedAt) < PRE_EVERY) return cur;
  return await buildPrematch();
}

/* 3.99.9 — BUDGET DELLE RICHIESTE TotalCorner (limite 30 al minuto). Ogni giro ne usa al massimo TC_RUN_MAX, in

const TC_RUN_MAX = 26;
function tcLeft() {

const tcNum = (v: any) => (v == null || v === "" ? null : isFinite(Number(v)) ? Number(v) : null);
/* 4.0.2 — LIMITE VERO DI TOTALCORNER: 30 richieste al minuto contate a finestre di 10 secondi, cioè al massimo 5
   richieste ogni 10 secondi (documentazione API). Il server le sparava tutte di fila: le pagine live, le quote e il
   prematch finivano in "troppe richieste" (prematch fermo, partite live perse). Ora si rispetta il ritmo: al massimo
   5 richieste in 10 secondi, e se TotalCorner dice che non ne restano si aspetta il suo "reset". */
const tcTimes: number[] = [];
let tcResetAt = 0;
async function tcThrottle() {
  const now = Date.now();
  while (tcTimes.length && now - tcTimes[0] > 10_500) tcTimes.shift();
  let wait = tcTimes.length >= 5 ? 10_500 - (now - tcTimes[0]) : 0;
  if (tcResetAt > now) wait = Math.max(wait, tcResetAt - now);
  if (wait > 0) await new Promise((r) => setTimeout(r, Math.min(wait, 11_000)));
  tcTimes.push(Date.now());
}
async function tcGet(path: string, params: Record<string, string | number> = {}) {
  await tcThrottle();
  tcCalls++;
  const qs = Object.entries(params).map(([k, v]) => k + "=" + encodeURIComponent(String(v))).join("&");
  const r = await fetch(TC + path + "?" + (qs ? qs + "&" : "") + "token=" + encodeURIComponent(TC_TOKEN), { headers: { Accept: "application/json" } });
  tcRem = r.headers.get("x-rate-limit-remaining") ?? tcRem;
  const rst = Number(r.headers.get("x-rate-limit-reset")), remN = Number(r.headers.get("x-rate-limit-remaining"));
  /* reset: secondi che mancano (o istante in secondi): se non restano richieste si aspetta fino a lì */
  if (isFinite(remN) && remN <= 0 && isFinite(rst) && rst > 0) tcResetAt = Date.now() + (rst > 1e9 ? Math.max(0, rst * 1000 - Date.now()) : rst * 1000) + 200;
  let j: any = null;
  try { j = await r.json(); } catch { throw new Error("TotalCorner " + r.status + ": risposta non valida"); }
  if (!r.ok || j?.success === 0 || j?.error) {
    const c = j?.error?.code || r.status, msg = j?.error?.message || "";
    if (c === "TOKEN_ERROR") throw new Error("TotalCorner: token non valido (segreto TOTALCORNER_TOKEN)");
    if (c === "NO_PERMISSION") throw new Error("TotalCorner: serve l'abbonamento VIP attivo");
    if (c === "TOO_MANY_REQUEST") throw new Error("TotalCorner: troppe richieste in un minuto (rateLimit)");
    throw new Error("TotalCorner " + c + (msg ? ": " + msg : ""));
  }
  return j;
}
async function tcList(path: string, params: Record<string, string | number>, maxPages = 8, firstPage = 1) {
  let out: any[] = [], page = firstPage, pages = firstPage;
  for (let n = 0; n < maxPages; n++) {
    const j = await tcGet(path, { ...params, page });
    const d = Array.isArray(j?.data) ? j.data : Array.isArray(j?.data?.data) ? j.data.data : [];
    out = out.concat(d);
    const pg = j?.pagination || j?.data?.pagination || {};
    pages = tcNum(pg.pages) ?? page;
    if (!pg.next || page >= pages || !d.length) { page++; break; }
    page++;
  }
  return { list: out, nextPage: page, pages };
}
function tcPair(x: any, ...keys: string[]): [number | null, number | null] {
  for (const k of keys) {
    const v = x?.[k];
    if (Array.isArray(v) && v.length >= 2) return [tcNum(v[0]), tcNum(v[1])];
    if (typeof v === "string" && v.includes(",")) { const p = v.split(","); return [tcNum(p[0]), tcNum(p[1])]; }
  }
  return [null, null];
}
/* linea gol ("2.5", "2.5,3.0", ["3.0"]) -> gol attesi della partita intera */
function tcLine(v: any): number | null {
  const s = Array.isArray(v) ? v[0] : v;
  if (s == null || s === "") return null;
  const parts = String(s).split(",").map(Number).filter((n) => isFinite(n));
  if (!parts.length) return null;
  return parts.reduce((a, b) => a + b, 0) / parts.length;
}
/* 3.89.0 — prepartita da sole quote TotalCorner: 68% vinti contro 80% previsto (66 pronostici): riduzione portata al 20% */
const TC_LINE_SCALE = 0.80;
function tcLamFromLine(L: number | null) {
  if (L == null || L <= 0) return null;
  const fr = L - Math.floor(L), adj = Math.abs(fr - 0.5) < 0.01 ? 0.17 : Math.abs(fr - 0.25) < 0.01 || Math.abs(fr - 0.75) < 0.01 ? 0.08 : 0;
  /* 3.64.5 — taratura sui risultati veri del 19/10: con la linea gol di TotalCorner i gol attesi erano 3,8
     di media contro 2,7 reali (19 partite). Si riduce del 15% finché non ci sono più dati. */
  return Math.max(0.6, Math.min(6, (L + adj) * TC_LINE_SCALE));
}
/* orario d'inizio: TotalCorner non dice il fuso; lo ricavo dalle partite live (inizio + minuto ≈ adesso) */
function tcParseStart(s: any) { const t = Date.parse(String(s || "").replace(" ", "T") + "Z"); return isFinite(t) ? t : null; }
function tcCalibrate(list: any[]) {
  const d: number[] = [];
  for (const x of list) {
    const t = tcParseStart(x.start), m = tcNum(x.status);
    if (t == null || m == null) continue;
    d.push((Date.now() - t - (m + (String(x.ish) === "1" ? 15 : 0)) * 60e3) / 3600e3);
  }
  if (d.length < 3) return;
  d.sort((a, b) => a - b);
  tcOff = Math.round(-d[Math.floor(d.length / 2)] * 4) / 4;
}
function tcStartUtc(s: any) { const t = tcParseStart(s); return t == null ? null : t - (tcOff || 0) * 3600e3; }
function tcStatus(x: any) {
  const raw = String(x.status ?? "").trim(), n = tcNum(raw);
  if (n != null) return { st: String(x.ish) === "1" || n > 46 ? "2H" : "1H", el: n };
  if (/half|ht|break/i.test(raw)) return { st: "HT", el: 45 };
  if (/^(?:full[ _-]?time|full|ft|ended|end|finished|finish)$/i.test(raw)) return { st: "FT", el: 90 };
  if (/post|canc|aband|susp/i.test(raw)) return { st: /post/i.test(raw) ? "PST" : "CANC", el: null };
  return { st: "NS", el: null };
}
function tcGoals(x: any) {
  return (Array.isArray(x.events) ? x.events : []).filter((e: any) => String(e.tp).toLowerCase() === "g")
    /* 4.2.3 — anche i gol nel recupero ("45+2" → 45, "90+1" → 90): prima si perdevano */
    .map((e: any) => { const s = String(e.t ?? "").replace(/['’]/g, "").trim(), m = /^(\d+)\s*\+/.exec(s); return { min: m ? Number(m[1]) : tcNum(s), side: e.h === "h" ? 0 : e.h === "a" ? 1 : null }; })
    .filter((g: any) => g.min != null && g.min >= 0).sort((a: any, b: any) => a.min - b.min);
}
function tcStats(x: any) {
  const on = tcPair(x, "shot_on", "shotOn"), off = tcPair(x, "shot_off", "shotOff"), pos = tcPair(x, "possess", "possession"),
    att = tcPair(x, "attacks"), da = tcPair(x, "dangerous_attacks", "dangerousAttacks", "dattacks", "dang_attacks");
  const hc = tcNum(x.hc), ac = tcNum(x.ac), hr = tcNum(x.hrc) ?? 0, ar = tcNum(x.arc) ?? 0;
  const has = on[0] != null || hc != null;
  if (!has) return null;
  const side = [0, 1].map((i) => ({
    s: on[i] != null || off[i] != null ? (on[i] || 0) + (off[i] || 0) : null, t: on[i], c: i ? ac : hc,
    poss: pos[i], red: i ? ar : hr, att: att[i], da: da[i], xg: null,
  }));
  return { s: (side[0].s || 0) + (side[1].s || 0), t: (side[0].t || 0) + (side[1].t || 0), c: hc != null && ac != null ? hc + ac : null, red: hr + ar, xg: null, side };
}

/* partite simulate (eSoccer, Cyber, virtuali, FIFA/eFootball con giocatori): da escludere */

}

function tipAsianLam(L, G, o, u) {
  if (!U.finite(L) || !(o > 1.01) || !(u > 1.01) || L - G <= 0) return null;
  var q = (1 / o) / (1 / o + 1 / u), f = 1 / q, fr = L - Math.floor(L);
  var halves = Math.abs(fr - 0.25) < 0.01 || Math.abs(fr - 0.75) < 0.01 ? [L - 0.25, L + 0.25] : [L];
  var cdf = function (k, l) { if (k < 0) return 0; var t = Math.exp(-l), c = t; for (var i = 1; i <= k; i++) { t *= l / i; c += t; } return c; };
  var g = function (l) {
    var s = 0;
    halves.forEach(function (h) {
      var need = h - G, pw, pp = 0;
      if (Math.abs(need - Math.round(need)) < 0.01) {
        var n = Math.round(need);
        pw = n >= 0 ? 1 - cdf(n, l) : 1;
        if (n >= 0) { var t = Math.exp(-l); for (var i = 1; i <= n; i++) t *= l / i; pp = t; }
      } else { var n2 = Math.floor(need); pw = n2 >= 0 ? 1 - cdf(n2, l) : 1; }
      s += pw * f + pp;
    });
    return s / halves.length - 1;
  };
  var lo = 1e-4, hi = 15;
  if (g(hi) < 0 || g(lo) > 0) return null;
  for (var it = 0; it < 60; it++) { var md = (lo + hi) / 2; if (g(md) < 0) lo = md; else hi = md; }
  return (lo + hi) / 2;
}

async function send(sub: any, payload: unknown) {
  if (!vapidOk) return "novapid";
  try {
    await webpush.sendNotification(
      { endpoint: sub.endpoint, keys: { p256dh: sub.p256dh, auth: sub.auth } },

    );

async function lgTrack(list: any[], wanted: string[] = []) {
  const seen: any = (await getState("lgSeenVerified")) || {}, now = Date.now();
  const ids = new Set<string>(), finished: any[] = [];
  for (const r of list || []) {
    if (!r?.afId || !r.league) continue;
    ids.add(String(r.afId));
    if (r.status === "FT") finished.push(r);
    else seen[r.afId] = { at: seen[r.afId]?.at || now, checkAt: seen[r.afId]?.checkAt || 0, min: r.elapsed, status: r.status };
  }
  // Disappearance only schedules a check. It never supplies a final score.
  /* 3.87.0 — prima le partite con segnali in attesa (finite ma non ancora nel registro), poi le altre sparite dal live */
  for (const id of wanted) if (!ids.has(id) && !seen[id]) seen[id] = { at: now, checkAt: 0 };
  const due = (id: string) => !ids.has(id) && id.startsWith("tc:") && now - (seen[id]?.checkAt || 0) >= 10 * 60e3;
  const want = new Set(wanted);
  const pending = Object.keys(seen).filter(due).sort((a, b) => (want.has(b) ? 1 : 0) - (want.has(a) ? 1 : 0) || seen[a].checkAt - seen[b].checkAt).slice(0, 6);
  for (const id of pending) {
    if (tcLeft() <= 0) break;
    seen[id].checkAt = now;
    try {
      const j = await tcGet("/match/view/" + id.slice(3), { columns: TC_COLS });
      const raw = Array.isArray(j?.data) ? j.data[0] : j?.data;
      if (!raw) continue;
      const r = tcFixture(raw);
      if (r.status === "FT" && r.resultConfirmed) finished.push(r);
      if (r.status === "CANC" || r.status === "PST") delete seen[id];
    } catch (e) { console.error("verifica finale", id, (e as Error).message); await tcBlock(e); break; }
  }
  const leagues = new Set<string>();
  const addedRows: any[] = [];
  let added = 0;
  for (const r of finished) {
    if (!r.resultConfirmed || !r.goals?.every((n: any) => Number.isInteger(n) && n >= 0)) continue;
    const sd = r.stats?.side || [{}, {}];
    const row = { af_id: r.afId, league: r.league, home: r.home || null, away: r.away || null,
      kickoff: r.kickoff, last_min: r.elapsed, goals: r.goals, ht: r.ht || null,
      corners: [sd[0]?.c ?? null, sd[1]?.c ?? null], ht_corners: r.htCorners || null, goal_list: Array.isArray(r.goalList) && r.goalList.length === r.goals[0] + r.goals[1] ? r.goalList : null, confirmed: true,
      confirmed_at: new Date(now).toISOString(), source: "totalcorner", model_version: "3.94.0" };
    let { error } = await admin.from("pronostici_finali").upsert(row, { onConflict: "af_id" });
    /* senza la migrazione 3.86 la colonna goal_list non c'è: si salva comunque il resto */
    if (error && /goal_list/.test(error.message)) { const { goal_list: _g, ...rest } = row as any; ({ error } = await admin.from("pronostici_finali").upsert(rest, { onConflict: "af_id" })); }
    if (error) { console.error("finali verificati", error.message); continue; }
    leagues.add(r.league); delete seen[r.afId]; added++; addedRows.push(row);
  }
  // Recompute from unique confirmed matches; repeated checks cannot inflate n.
  const avg: any = (await getState("lgAvgVerified")) || {};
  for (const league of leagues) {
    const { data, error } = await admin.from("pronostici_finali").select("goals")
      .eq("confirmed", true).eq("league", league).order("kickoff", { ascending: false }).limit(200);
    if (error) { console.error("medie verificate", error.message); continue; }
    const valid = (data || []).filter((r: any) => Array.isArray(r.goals) && r.goals.length === 2 && r.goals.every((v: any) => Number.isInteger(v) && v >= 0));
    avg[league] = { n: valid.length, g: valid.reduce((n: number, r: any) => n + r.goals[0] + r.goals[1], 0), t: now };
  }
  for (const id of Object.keys(seen)) if (now - seen[id].at > 7 * 86400e3) delete seen[id];
  await setState("lgSeenVerified", seen);
  await setState("lgAvgVerified", avg);
  lgAvgCache = avg;
  try { await lateCorAdd(addedRows); } catch (e) { console.error("angoli finali", (e as Error).message); }
  return added;
}

/* 3.99.4 — QUOTE GOL LIVE (bet365 via TotalCorner) per le partite vicine a un segnale gol: 1T 10’-38’ con mercato
   ≥63%, 2T 46’-80’ con mercato ≥70% (dalla sola linea). Al massimo 6 richieste per giro, tenute 100 secondi.
   L’ultima riga di goal_list / goal_half_list = { min, line, over, under }. Viaggia con la partita (x.goalOdds). */
async function goalOddsEnrich(list: any[]) {
  const cand = (list || []).filter((r: any) => typeof r?.afId === "string" && r.afId.startsWith("tc:") && r.stats && r.liveLine != null && Array.isArray(r.goals)).map((r: any) => {
    const half = r.status === "1H" ? 1 : r.status === "2H" ? 2 : 0, G = (r.goals[0] || 0) + (r.goals[1] || 0);
    const ok = half === 1 ? r.elapsed >= 10 && r.elapsed <= 38 : half === 2 ? r.elapsed >= 46 && r.elapsed <= 80 : false;
    const M = ok ? tipMktNext(r, r.elapsed, half, G) : null;
    /* 4.2.1 — anche 1-0 / 0-1 a inizio 2T con linea bassa (nuovo tv2): quota vera da registrare */
    return M && M.p >= (half === 1 ? 0.63 : G === 1 && r.elapsed <= 52 ? 0.60 : 0.70) ? { r, p: M.p } : null;
  }).filter(Boolean).sort((a: any, b: any) => b.p - a.p);
  /* 4.6.0 — QUOTA RAGGIUNTA: le partite seguite dopo un segnale (stato qWatch, scritto dal giro prima) vicine alla soglia
     dell'utente hanno la quota vera a ogni giro (fino a 5 in più), così l'avviso parte sulla quota bet365 e non sulla linea */
  const W: any = (await getState("qWatch")) || {}, nowW = Date.now();
  const wc = (list || []).map((r: any) => {
    const w = W[r?.afId];
    if (!w || nowW - w.at > 6 * 60e3 || !Array.isArray(r.goals) || r.liveLine == null) return null;
    const half = r.status === "1H" ? 1 : r.status === "2H" ? 2 : 0, G = (r.goals[0] || 0) + (r.goals[1] || 0);
    if (half !== w.half || G !== w.G) return null;
    const M = tipMktNext(r, r.elapsed, half, G), q = M && M.p > 0 ? 0.95 / M.p : null;
    return q && q >= w.Q - 0.35 ? { r, d: Math.abs(q - w.Q) } : null;
  }).filter(Boolean).sort((a: any, b: any) => a.d - b.d);
  if (!cand.length && !wc.length) return 0;
  const cache: any = (await getState("goalOdds")) || {}, now = Date.now();
  const last = (L: any) => {
    const rows = (Array.isArray(L) ? L : []).filter((x: any) => Array.isArray(x) && x.length >= 4);
    if (!rows.length) return null;
    rows.sort((a: any, b: any) => (tcNum(a[0]) ?? -1) - (tcNum(b[0]) ?? -1) || String(a[4] ?? "").localeCompare(String(b[4] ?? "")));
    /* riga TotalCorner: [minuto, linea, over, under, ora, gol casa, gol ospite] */
    const x = rows[rows.length - 1], o: any = { min: tcNum(x[0]), line: tcLine(x[1]), over: tcNum(x[2]), under: tcNum(x[3]) };
    const gh = tcNum(x[5]), ga = tcNum(x[6]);
    if (gh != null && ga != null) o.g = [gh, ga];
    return o.line != null ? o : null;
  };
  let asked = 0, askedW = 0;
  for (const { r, w } of [...cand.map((c: any) => ({ r: c.r, w: false })), ...wc.map((c: any) => ({ r: c.r, w: true }))]) {
    const c = cache[r.afId];
    if (c && now - c.at < 100e3) { r.goalOdds = c.v; continue; }
    if ((w ? askedW >= 5 : asked >= 6) || tcLeft() <= 0) continue; /* 4.2.8 — 6 partite a giro (erano 4): le % dei segnali vengono dalle quote vere */
    if (w) askedW++; else asked++;
    try {
      const j = await tcGet("/match/odds/" + r.afId.slice(3), { columns: "goalList,goalHalfList" });
      const d = Array.isArray(j?.data) ? j.data[0] : j?.data;
      const val = { ft: last(d?.goal_list ?? d?.goalList), ht: last(d?.goal_half_list ?? d?.goalHalfList) };
      cache[r.afId] = { at: now, v: val }; r.goalOdds = val;
      if (!val.ft && !val.ht) await setState("goalOddsRaw", { at: now, id: r.afId, keys: Object.keys(d || {}), sample: JSON.stringify(j).slice(0, 1500) });
    } catch (e) { console.error("quote gol", r.afId, (e as Error).message); break; }
  }
  for (const k of Object.keys(cache)) if (now - cache[k].at > 3 * 3600e3) delete cache[k];
  await setState("goalOdds", cache);
  return asked + askedW;
}

/* 3.99.6 — STORICO QUOTE TOTALCORNER PER IL BACKTEST (tabella pronostici_storico_quote, migrazione-3.99.6.sql).
   Si va indietro di HIST_DAYS giorni a partire da ieri: per ogni giorno l'elenco delle partite (/match/schedule, una

   gol casa, gol ospite], una per minuto (l'ultima). */
const HIST_DAYS = 30;
function histDayBack(d: string) { const t = Date.parse(d + "T12:00:00Z") - 86400e3; return new Date(t).toISOString().slice(0, 10); }
function histList(L: any) {
  const rows = (Array.isArray(L) ? L : []).filter((x: any) => Array.isArray(x) && x.length >= 4);
  const byMin = new Map<number, any[]>();
  for (const x of rows) {
    const raw = String(x[0] ?? "").trim(), m = tcNum(raw) ?? (/half|ht/i.test(raw) ? 45.5 : -1); /* -1 = prepartita */
    const line = tcLine(x[1]), o = tcNum(x[2]), u = tcNum(x[3]);
    if (m == null || line == null) continue;
    byMin.set(m, [m, line, o, u, tcNum(x[5]), tcNum(x[6])]);
  }
  return [...byMin.values()].sort((a, b) => a[0] - b[0]);
}
function histFirst(L: any) {
  const rows = (Array.isArray(L) ? L : []).filter((x: any) => Array.isArray(x) && x.length >= 4);
  if (!rows.length) return null;
  const x = rows[0];
  return [String(x[0] ?? ""), String(x[1] ?? ""), tcNum(x[2]), tcNum(x[3])];
}
/* 4.2.2 — riga PREPARTITA (le liste arrivano dalla più recente: prima si salvava quella di fine partita) */
function histPre(L: any) {
  const rows = (Array.isArray(L) ? L : []).filter((x: any) => Array.isArray(x) && x.length >= 4);
  if (!rows.length) return null;
  const pre = rows.filter((x: any) => { const m = String(x[0] ?? "").trim(); return tcNum(m) == null && !/half|ht/i.test(m); });
  const x = pre.length ? pre[pre.length - 1] : rows[rows.length - 1];
  return [String(x[0] ?? ""), String(x[1] ?? ""), tcNum(x[2]), tcNum(x[3])];
}
function histRow(d: any) {
  if (!d || d.id == null) return null;
  const hg = tcNum(d.hg), ag = tcNum(d.ag);
  if (hg == null || ag == null) return null;
  const pair = (a: any, b: any) => { const x = tcNum(a), y = tcNum(b); return x != null && y != null ? [x, y] : null; };
  const gl = histList(d.goal_list ?? d.goalList), cl = histList(d.corner_list ?? d.cornerList);
  const ko = tcStartUtc(d.start);
  return { af_id: "tc:" + d.id, league: String(d.l || ""), home: String(d.h || ""), away: String(d.a || ""),
    kickoff: ko != null ? new Date(ko).toISOString() : null, goals: [hg, ag], ht: pair(d.hf_hg, d.hf_ag), corners: pair(d.hc, d.ac),
    ht_corners: pair(d.hf_hc, d.hf_ac), red: pair(d.hrc, d.arc), gl, glh: histList(d.goal_half_list ?? d.goalHalfList),
    cl, clh: histList(d.corner_half_list ?? d.cornerHalfList), x12: histPre(d.odds_list ?? d.oddsList), ah: histPre(d.asian_list ?? d.asianList) };
}
async function histBackfill() {
  if (!TC_TOKEN || (await tcBlocked())) return null;
  const chk = await admin.from("pronostici_storico_quote").select("af_id", { head: true, count: "exact" }).limit(1);
  if (chk.error) return { error: "manca la tabella pronostici_storico_quote: esegui migrazione-3.99.6.sql" };
  const now = Date.now();
  const st: any = (await getState("histBF")) || {};
  if (!st.start) Object.assign(st, { start: romeParts(now - 86400e3).date, stop: romeParts(now - HIST_DAYS * 86400e3).date, day: null, page: 1, queue: [], dayEnd: false, n: 0, skip: 0, err: 0, days: {} });
  if (st.done) return { done: true, n: st.n };
  let budget = Math.min(6, tcLeft());
  if (budget <= 0) return { wait: true, n: st.n };
  if (!st.day) st.day = st.start;
  while (budget > 0 && !st.done) {
    if (st.queue.length) {
      const id = st.queue.shift(); budget--;
      try {
        const j = await tcGet("/match/odds/" + id, { columns: "goalList,goalHalfList,cornerList,cornerHalfList,oddsList,asianList" });
        const d = Array.isArray(j?.data) ? j.data[0] : j?.data;

        const row = histRow(d);
        if (!row || (!row.gl.length && !row.cl.length)) { st.skip++; continue; }
        const { error } = await admin.from("pronostici_storico_quote").upsert(row, { onConflict: "af_id" });
        if (error) { st.err++; st.lastErr = error.message; if (/relation|does not exist/i.test(error.message)) { st.queue.unshift(id); break; } continue; }
        st.n++; st.days[st.day] = (st.days[st.day] || 0) + 1;
      } catch (e) { st.err++; st.lastErr = (e as Error).message; if (/rateLimit|troppe/.test(st.lastErr)) { st.queue.unshift(id); break; } }
      continue;
    }
    if (st.dayEnd) {
      if (st.day <= st.stop) { st.done = true; break; }
      st.day = histDayBack(st.day); st.page = 1; st.dayEnd = false; continue;
    }
    budget--;
    const r = await tcList("/match/schedule", { date: st.day.replace(/-/g, ""), columns: "goalLine" }, 1, st.page);
    const real = r.list.filter((x: any) => !tcIsSim(x) && x?.id != null);
    const withLine = real.filter((x: any) => x.p_goal != null && x.p_goal !== "" && !(Array.isArray(x.p_goal) && !x.p_goal.length));
    st.queue = (real.some((x: any) => "p_goal" in x) ? withLine : real).map((x: any) => String(x.id));
    if (r.nextPage > r.pages || !r.list.length) st.dayEnd = true; else st.page = r.nextPage;
  }
  st.at = now;
  await setState("histBF", st);
  return { day: st.day, page: st.page, queue: st.queue.length, n: st.n, skip: st.skip, err: st.err, done: !!st.done };
}

/* 3.99.0 — STORICO ANGOLI FINALI PER CAMPIONATO E PAESE. A ogni finale verificato si guarda nel registro
   (pronostici_traccia) la prima istantanea tra il 34’ e il 36’ e tra l’82’ e l’84’: c’è stato almeno un angolo in più
   entro la fine del tempo? Conteggi n1/w1 (1° tempo) e n2/w2 (finale) per campionato e per Paese (prima parola del
   nome). Ogni partita si conta una volta sola. Il motore (tipLateCor) li usa da 8 partite in su. */
async function lateCorAdd(rows: any[]) {
  const okC = (c: any) => Array.isArray(c) && c.length === 2 && c.every((n: any) => Number.isInteger(n) && n >= 0);
  const st: any = (await getState("lateCor")) || { base: { 1: 0.72, 2: 0.67 }, lg: {}, cty: {}, done: {} };
  st.lg = st.lg || {}; st.cty = st.cty || {}; st.done = st.done || {};
  const todo = (rows || []).filter((r: any) => r && r.af_id && !st.done[r.af_id] && okC(r.corners)).slice(0, 40);
  if (!todo.length) return 0;
  const { data, error } = await admin.from("pronostici_traccia").select("af_id,min,st,d,created_at")
    .in("af_id", todo.map((r: any) => r.af_id))
    .or("and(st.eq.1H,min.gte.34,min.lte.36),and(st.eq.2H,min.gte.82,min.lte.84)")
    .order("created_at", { ascending: true }).limit(2000);
  if (error) throw new Error(error.message);
  const cOf = (d: any) => d && okC(d.c) ? d.c[0] + d.c[1] : null;
  const bump = (o: any, key: string, h: number, y: number) => { const t = (o[key] = o[key] || {}); t["n" + h] = (t["n" + h] || 0) + 1; t["w" + h] = (t["w" + h] || 0) + y; };
  let n = 0;
  for (const r of todo) {
    const snaps = (data || []).filter((t: any) => t.af_id === r.af_id);
    const s1 = snaps.find((t: any) => t.st === "1H"), s2 = snaps.find((t: any) => t.st === "2H");
    const cty = String(r.league || "").split(" ")[0];
    const c1 = s1 ? cOf(s1.d) : null, c2 = s2 ? cOf(s2.d) : null;
    const red = (t: any) => Array.isArray(t?.d?.red) && t.d.red.some((v: any) => v > 0);
    if (c1 != null && okC(r.ht_corners) && !red(s1) && r.ht_corners[0] + r.ht_corners[1] >= c1) {
      const y = r.ht_corners[0] + r.ht_corners[1] > c1 ? 1 : 0; bump(st.lg, r.league, 1, y); if (cty) bump(st.cty, cty, 1, y); n++;
    }
    if (c2 != null && !red(s2) && r.corners[0] + r.corners[1] >= c2) {
      const y = r.corners[0] + r.corners[1] > c2 ? 1 : 0; bump(st.lg, r.league, 2, y); if (cty) bump(st.cty, cty, 2, y); n++;
    }
    st.done[r.af_id] = Date.now();
  }
  for (const k of Object.keys(st.done)) if (Date.now() - st.done[k] > 10 * 86400e3) delete st.done[k];
  st.t = Date.now();
  await setState("lateCor", st);
  return n;
}

/* 3.71.0 — ANGOLI: per le partite nelle finestre (1T 33’-43’, 2T 79’-88’) storico delle squadre dai finali verificati
   (angoli fatti/subiti, ultime 10, anche all'intervallo) e quote angoli live bet365 (al massimo 6 richieste per giro,
   tenute 100 secondi). Tutto viaggia con la partita: lo usano sia gli avvisi sia l'app. */
async function cornerEnrich(list: any[]) {
  const cand = (list || []).filter((r: any) => typeof r?.afId === "string" && r.afId.startsWith("tc:") && r.stats &&
    ((r.status === "1H" && r.elapsed >= 33 && r.elapsed <= 43) || (r.status === "2H" && r.elapsed >= 79 && r.elapsed <= 89)));
  if (!cand.length) return 0;
  const ok = (c: any) => Array.isArray(c) && c.length === 2 && c.every((n: any) => Number.isInteger(n) && n >= 0);
  const leagues = [...new Set(cand.map((r: any) => r.league).filter(Boolean))];
  try {
    const { data, error } = await admin.from("pronostici_finali").select("league,home,away,kickoff,corners,ht_corners")
      .eq("confirmed", true).in("league", leagues).order("kickoff", { ascending: false }).limit(3000);
    if (error) throw new Error(error.message);
    const rows = (data || []).filter((r: any) => ok(r.corners));
    const team = (league: string, name: string) => {
      const L = rows.filter((r: any) => r.league === league && (r.home === name || r.away === name)).slice(0, 10);
      if (!L.length) return null;
      let cf = 0, ca = 0, ht = 0, ft = 0, nHt = 0;
      for (const r of L) {
        const i = r.home === name ? 0 : 1;
        cf += r.corners[i]; ca += r.corners[1 - i];
        if (ok(r.ht_corners)) { ht += r.ht_corners[0] + r.ht_corners[1]; ft += r.corners[0] + r.corners[1]; nHt++; }
      }
      return { n: L.length, cf: cf / L.length, ca: ca / L.length, nHt, htShare: nHt && ft ? ht / ft : null };
    };
    for (const r of cand) {
      const lg = rows.filter((z: any) => z.league === r.league).slice(0, 200), lgHt = lg.filter((z: any) => ok(z.ht_corners));
      r.cornerHist = { h: team(r.league, r.home), a: team(r.league, r.away),
        lg: lg.length ? { n: lg.length, avg: lg.reduce((n: number, z: any) => n + z.corners[0] + z.corners[1], 0) / lg.length,
          htAvg: lgHt.length >= 5 ? lgHt.reduce((n: number, z: any) => n + z.ht_corners[0] + z.ht_corners[1], 0) / lgHt.length : null } : null };
    }
  } catch (e) { console.error("storico angoli", (e as Error).message); }
  const cache: any = (await getState("cornerOdds")) || {}, now = Date.now();
  const last = (L: any) => {
    const rows = (Array.isArray(L) ? L : []).filter((x: any) => Array.isArray(x) && x.length >= 4);
    if (!rows.length) return null;
    rows.sort((a: any, b: any) => (tcNum(a[0]) ?? -1) - (tcNum(b[0]) ?? -1) || String(a[4] ?? "").localeCompare(String(b[4] ?? "")));
    const x = rows[rows.length - 1], o = { min: tcNum(x[0]), line: tcLine(x[1]), over: tcNum(x[2]), under: tcNum(x[3]) };
    return o.line != null ? o : null;
  };
  let asked = 0;
  for (const r of cand.sort((a: any, b: any) => (b.elapsed || 0) - (a.elapsed || 0))) {
    const c = cache[r.afId];
    if (c && now - c.at < 100e3) { r.cornerOdds = c.v; continue; }
    if (asked >= 2 || tcLeft() <= 0) continue; /* 4.2.8 — angoli solo nello storico: 2 a giro (erano 4), il resto alle quote gol */
    asked++;
    try {
      const j = await tcGet("/match/odds/" + r.afId.slice(3), { columns: "cornerList,cornerHalfList" });
      const d = Array.isArray(j?.data) ? j.data[0] : j?.data;
      const val = { ft: last(d?.corner_list ?? d?.cornerList), ht: last(d?.corner_half_list ?? d?.cornerHalfList) };
      cache[r.afId] = { at: now, v: val }; r.cornerOdds = val;
      if (!val.ft && !val.ht) await setState("cornerOddsRaw", { at: now, id: r.afId, keys: Object.keys(d || {}), sample: JSON.stringify(j).slice(0, 1500) });
    } catch (e) { console.error("quote angoli", r.afId, (e as Error).message); break; }
  }
  for (const k of Object.keys(cache)) if (now - cache[k].at > 3 * 3600e3) delete cache[k];
  await setState("cornerOdds", cache);
  return asked;
}
function tipPlayOutcome(e, x) {
  if (!Array.isArray(x.ft)) return null;
  if (x.ft[0] < e.gh || x.ft[1] < e.ga || (e.half === 1 && Array.isArray(x.ht) && (x.ht[0] < e.gh || x.ht[1] < e.ga))) return null;
  var G = e.gh + e.ga, FT = x.ft[0] + x.ft[1], HT = Array.isArray(x.ht) ? x.ht[0] + x.ht[1] : null,
    mins = Array.isArray(x.goals) && x.goals.length === FT ? x.goals.map(function (g) { return g.min; }) : null; /* 3.90.0 — solo elenco completo */
  if (e.k === 'tc1' || e.k === 'tc2' || e.k === 'tk2') {
    if (!Array.isArray(x.corners) || x.corners[0] == null || x.corners[1] == null || e.cn == null) return null;
    if (!x.corners.every(function (n) { return Number.isInteger(n) && n >= 0; }) || x.corners[0] + x.corners[1] < e.cn) return null;
    return x.corners[0] + x.corners[1] - e.cn >= (e.need || (e.k === 'tc2' ? 2 : 1)) ? 1 : 0;
  }
  if (e.k === 'th1' || e.k === 'th2' || e.k === 'tk1') {
    var hc = x.htCorners;
    if (!Array.isArray(hc) || hc.length !== 2 || !hc.every(function (n) { return Number.isInteger(n) && n >= 0; }) || e.cn == null || hc[0] + hc[1] < e.cn) return null;
    return hc[0] + hc[1] - e.cn >= (e.need || (e.k === 'th2' ? 2 : 1)) ? 1 : 0;
  }
  if (e.k === 'tp1' || e.k === 'tq1' || e.k === 'tg1' || e.k === 'tf1' || e.k === 'tm1' || e.k === 'tm1s' || e.k === 'tr1') return HT == null ? null : HT > G ? 1 : 0; /* 4.6.0 — tr1: quota raggiunta 1T */
  if (e.k === 'tq2' || e.k === 'tn2') return FT > G ? 1 : 0;
  if (e.k === 'tp2') return FT > G ? 1 : 0;
  if (e.k === 't2w' || e.k === 't1w') {
    var tot = e.k === 't1w' ? HT : FT;
    if (tot == null) return null;
    if (tot === G) return 0;
    if (!mins) return null;
    if (mins.some(function (mm) { return mm > e.min && mm <= e.at; })) return 'skip';
    return mins.some(function (mm) { return mm > e.at && mm <= (e.k === 't1w' ? 45 : 200); }) ? 1 : 0;
  }
  if (e.k.indexOf('t1') === 0) return HT == null ? null : HT > G ? 1 : 0;
  return FT > G ? 1 : 0;
}
/* 3.87.0 — ESITI CHIUSI DAL SERVER: a ogni giro i segnali in attesa degli ultimi 3 giorni vengono confrontati con il

}

      const G = (y.x.goals[0] || 0) + (y.x.goals[1] || 0);
      if (best.green && (best.k === "t2u15" || best.k === "t2u18") && G > 0) {
        if (!oddsCache.has(x.afId)) oddsCache.set(x.afId, await bookN1(x.afId, G));
        const q = oddsCache.get(x.afId);
        if (q) { plays = tipPlays(y.r, y.min, y.half, { n1: q }); best = tipPlayBest(plays); if (!best) continue; }
      }
      const ctx = { min: y.min, half: y.half, gh: y.x.goals[0] || 0, ga: y.x.goals[1] || 0 };
      let e = tipPlayEntry(y.m, ctx, best, x.afId);
      e.src = "server";
      e.t = now;
      const was = known.get(e.key);
      const upgraded = was === "try" && e.tier === "excellent";
      if (was && !upgraded) continue;
      if (originals.get(e.key)) e = tipSignalRevision(originals.get(e.key), e);
      /* 3.73.0 — filtri degli avvisi; se la giocata migliore è un tipo spento, si avvisa per un'altra giocata attiva della stessa partita */
      const whyE = tipAlertWhy(u.prefs, e);
      let pushE: any = !whyE ? e : null, pushX: any = best;
      if (!pushE) {
        /* 3.91.0 — l'alternativa si controlla come segnale completo (probabilità e campionato): prima mancavano e
           "Solo qualità" la scartava sempre */
        for (const p of plays) {
          if (p === best || !(p.green || p.warm)) continue;
          const cand = tipPlayEntry(y.m, ctx, p, x.afId);
          if (tipAlertWants(u.prefs, cand)) { pushE = cand; pushE.t = now; pushX = p; break; }
        }
      }
      const wantPush = !!pushE;
      e.push = { sent: wantPush, why: wantPush ? (pushE === e ? "" : "notificato come " + (TIP_PLAY_SHORT[pushX.k] || pushX.k)) : whyE, at: now };
      const row = { user_id: u.id, key: e.key, tier: e.tier, data: e, updated_at: new Date().toISOString(), pushed: wantPush ? pushE.tier : null };
      const { error: insErr } = await admin.from("pronostici_segnali").upsert(row, { onConflict: "user_id,key" });
      if (insErr) { console.error("segnali", insErr.message); continue; }
      known.set(e.key, e.tier);
      newSig++;
      if (wantPush) { const n = await pushAll(u, message(pushE, pushX, upgraded, u.prefs), e); sent += n; }
    }
    /* 4.6.0 — quota raggiunta dopo i segnali delle ultime ore */
    try {
      const qr = await quotaWatch(u, list, now, known, originals);
      sent += qr.sent; newSig += qr.created; qNew += qr.created;
      for (const w of qr.watch) qWatch[w.afId] = { half: w.half, G: w.G, Q: Math.min(w.Q, qWatch[w.afId]?.Q ?? 9), at: now };
    } catch (e) { console.error("quota raggiunta", (e as Error).message); }
    /* storico delle scansioni (serve al modello completo dopo 5 minuti) */
    if (snaps.length) {
      tipLiveHistMany(snaps);
      try { await setState("hist:" + u.id, JSON.parse(__ls["tipLiveHist"] || "{}")); } catch { /* ok */ }
    }
  }
  /* pulizia: segnali più vecchi di 10 giorni */
  if (new Date(now).getUTCMinutes() < 3) await admin.from("pronostici_segnali").delete().lt("created_at", new Date(now - 60 * 86400e3).toISOString()); /* 3.99.0 — 60 giorni (prima 10): servono all'export */
  try { await setState("qWatch", qWatch); } catch { /* ok */ }
  return { users: users.length, live: list.length, evaluated, newSignals: newSig, quotaReached: qNew, sent, bzCalls };
}

    if (!Number.isFinite(from) || !Number.isFinite(until) || until < from || until - from > 45 * 86400e3)
